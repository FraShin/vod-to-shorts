# ============================================================
#  VOD -> Shorts pipeline  |  Python dependencies
# ============================================================
#  Tested on: Python 3.11.9 on Windows 11 and Python 3.12.3 on Linux/WSL2,
#  both with an NVIDIA GPU (CUDA 12.1).
#  Required Python: 3.11+
#
#  Install:
#      python -m venv .venv
#      source .venv/bin/activate        # Windows: .venv\Scripts\activate
#      pip install -r requirements.txt
#
#  On Windows, if activating the venv fails with an execution-policy error, call
#  the interpreter by path instead. It installs into the right place either way:
#      .venv\Scripts\python.exe -m pip install -r requirements.txt
#
#  WARNING ABOUT TORCH: the default PyPI `torch` package is the CPU build.
#  A GPU needs the CUDA wheels. If you have an NVIDIA card, install torch and
#  torchaudio from the PyTorch index first, then the rest:
#      pip install torch==2.5.1 torchaudio==2.5.1 --index-url https://download.pytorch.org/whl/cu121
#      pip install -r requirements.txt
#  Without a GPU the pipeline still runs: transcribe_engine detects the absence
#  of CUDA and falls back to CPU (much slower).
# ============================================================

# --- Core pipeline ---
faster-whisper==1.2.1
torch==2.5.1
numpy==2.4.4
soundfile==0.13.1
requests==2.34.0
openai==2.36.0

# --- CUDA libraries distributed through pip ---
# transcribe_engine.py makes these visible to CTranslate2/faster-whisper: on
# Linux/WSL through LD_LIBRARY_PATH, on Windows through os.add_dll_directory.
# win_amd64 wheels exist too, so they work on both.
# With a system CUDA install (or without an NVIDIA card) you can drop both:
# the import is guarded and the pipeline falls back to CPU.
nvidia-cublas-cu12==12.1.3.1
nvidia-cudnn-cu12==9.1.0.70

# --- Only for stable_ts_probe.py (A/B comparison, not part of the pipeline) ---
stable-ts==2.19.1
# stable-ts depends on torchaudio without pinning it, so pip installs the newest
# release, which is built against the newest torch. Its compiled DLLs then refuse
# to load against the torch pinned above -- `OSError [WinError 127]` at import --
# while `pip check` still reports "No broken requirements found", because the
# mismatch is in the binaries and not in the metadata. Pinning the pair to the
# same version is what keeps it loadable. With a GPU, install this one from the
# PyTorch cu121 index as well (see the header).
torchaudio==2.5.1
